$found = $false

function Suspicious($title, $detail) {
    $script:found = $true
    Write-Host "`n[SUSPICIOUS] $title" -ForegroundColor Red
    Write-Host $detail -ForegroundColor Red
}

# 1. Running kernel drivers
Get-CimInstance Win32_SystemDriver |
Where-Object State -eq "Running" |
ForEach-Object {

    $name = $_.Name
    $path = $_.PathName -replace '"',''

    if ($path -and (Test-Path $path)) {
        $sig = Get-AuthenticodeSignature $path

        if ($sig.Status -ne "Valid") {
            Suspicious "Driver: $name" "Path: $path | Signature: $($sig.Status)"
        }

        if ($path -match '(?i)\\Temp\\|\\AppData\\|\\Downloads\\|\\Users\\Public\\') {
            Suspicious "Driver: $name" "Unusual path: $path"
        }
    }
}

# 2. Suspicious process locations / signatures
Get-Process |
ForEach-Object {
    try {
        $p = $_
        $path = $p.Path

        if ($path) {
            $sig = Get-AuthenticodeSignature $path

            if ($sig.Status -ne "Valid") {
                Suspicious "Process: $($p.ProcessName) PID=$($p.Id)" `
                    "Signature: $($sig.Status) | Path: $path"
            }

            if ($path -match '(?i)\\Temp\\|\\AppData\\Local\\Temp\\|\\Downloads\\|\\Users\\Public\\') {
                Suspicious "Process: $($p.ProcessName) PID=$($p.Id)" `
                    "Unusual path: $path"
            }
        }
    } catch {}
}

# 3. Code Integrity events
Get-WinEvent -FilterHashtable @{
    LogName = 'Microsoft-Windows-CodeIntegrity/Operational'
} -MaxEvents 300 -ErrorAction SilentlyContinue |
Where-Object {
    $_.Message -match '(?i)blocked|unsigned|failed|violation|code integrity'
} |
ForEach-Object {
    Suspicious "Code Integrity Event" `
        "Time: $($_.TimeCreated)`n$($_.Message)"
}

# 4. Recent driver/service installation events
Get-WinEvent -FilterHashtable @{
    LogName = 'System'
    Id      = 7045
} -MaxEvents 100 -ErrorAction SilentlyContinue |
ForEach-Object {
    if ($_.Message -match '(?i)\.sys|driver|kernel') {
        Suspicious "Recent Driver Installation" `
            "Time: $($_.TimeCreated)`n$($_.Message)"
    }
}

if (-not $found) {
    Write-Host "`n[OK] No suspicious indicators found." -ForegroundColor Green
}
